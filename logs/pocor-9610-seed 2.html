Developer API Reference

All requests require Authorization: Bearer <token> and X-API-KEY: apikeytest. Obtain token via the login call below.

POST /api/v4/login — Authenticate 
curl -sk -X POST https://localhost:8482/core/api/v4/login \
  -H "Content-Type: application/json" \
  -d '{
    "username": "admin",
    "password": "demo",
    "api_key": "apikeytest"
  }'

# Response: { "token": "eyJ..." }
GET /api/v5/institutions — List institutions 
curl -sk https://localhost:8482/core/api/v5/institutions?limit=500 \
  -H "Authorization: Bearer $TOKEN" \
  -H "X-API-KEY: apikeytest"

# Response shape: { "data": { "current_page": 1, "data": [ { "id": 6, "name": "..." } ] } }
GET /api/v5/institution-registrations — List registrations 
curl -sk "https://localhost:8482/core/api/v5/institution-registrations?limit=200" \
  -H "Authorization: Bearer $TOKEN" \
  -H "X-API-KEY: apikeytest"
POST /api/v5/institution-registrations — Create registration 
curl -sk -X POST https://localhost:8482/core/api/v5/institution-registrations \
  -H "Authorization: Bearer $TOKEN" \
  -H "X-API-KEY: apikeytest" \
  -H "Content-Type: application/json" \
  -d '{
    "institution_id": 6,
    "valid_from": "2024-01-01",
    "valid_to": "2026-12-31"
  }'

# Response: { "data": { "id": 1, "institution_id": 6, "valid_from": "2024-01-01", "valid_to": "2026-12-31" } }
DELETE /api/v5/institution-registrations/{id} — Delete registration 
curl -sk -X DELETE https://localhost:8482/core/api/v5/institution-registrations/1 \
  -H "Authorization: Bearer $TOKEN" \
  -H "X-API-KEY: apikeytest"

# Response: 204 No Content
GET /api/v5/institution-accreditations — List accreditations 
curl -sk "https://localhost:8482/core/api/v5/institution-accreditations?limit=200" \
  -H "Authorization: Bearer $TOKEN" \
  -H "X-API-KEY: apikeytest"
POST /api/v5/institution-accreditations — Create accreditation 
curl -sk -X POST https://localhost:8482/core/api/v5/institution-accreditations \
  -H "Authorization: Bearer $TOKEN" \
  -H "X-API-KEY: apikeytest" \
  -H "Content-Type: application/json" \
  -d '{
    "institution_id": 6,
    "education_programme_id": 12,
    "valid_from": "2024-01-01",
    "valid_to": "2026-12-31"
  }'

# education_programme_id comes from /api/v5/education-programmes
# chain: education_programmes → education_cycles.education_level_id
#        → education_levels.education_system_id
#        → education_systems.academic_period_id
DELETE /api/v5/institution-accreditations/{id} — Delete accreditation 
curl -sk -X DELETE https://localhost:8482/core/api/v5/institution-accreditations/1 \
  -H "Authorization: Bearer $TOKEN" \
  -H "X-API-KEY: apikeytest"
GET /api/v5/education-programmes — List programmes (for accreditation FK) 
curl -sk "https://localhost:8482/core/api/v5/education-programmes?limit=500" \
  -H "Authorization: Bearer $TOKEN" \
  -H "X-API-KEY: apikeytest"
